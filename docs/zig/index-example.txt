const ctoon = @import("ctoon");

pub fn main() !void {
    const gpa = std.heap.page_allocator;

    // Decode ← TOON
    var val = try ctoon.loads(gpa, "name: Alice\nage: 30");
    defer val.deinit(gpa);
    std.debug.print("{s}\n", .{val.get("name").?.asStr().?}); // Alice

    // Encode → TOON
    const toon = try ctoon.dumps(gpa, val);
    defer gpa.free(toon);

    // JSON interop
    const json = try ctoon.dumpsJson(gpa, val, 2);
    defer gpa.free(json);
    var back = try ctoon.loadsJson(gpa, json);
    defer back.deinit(gpa);
    std.debug.print("{}\n", .{back.get("age").?.asU64().?}); // 30
}</code></pre>
